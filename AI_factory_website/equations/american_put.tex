\begin{equation*}
\begin{aligned}
C(t,S_t)
  &= \mathbb{E}^{\mathbb{Q}}_t\!\left[
      e^{-r\Delta t}V(t+\Delta t,S_{t+\Delta t})
    \right], \\[0.5em]
V(t,S_t)
  &= \max\!\left\{(K-S_t)^{+},\;C(t,S_t)\right\}, \\[0.5em]
V(T,S_T) &= (K-S_T)^{+}.
\end{aligned}
\end{equation*}
