\begin{equation*}
\begin{aligned}
r_t &= x_t+y_t, \\[0.5em]
\mathrm{d}x_t
  &= -a x_t\,\mathrm{d}t + \sigma\,\mathrm{d}W_t^x, \\[0.5em]
\mathrm{d}y_t
  &= -b y_t\,\mathrm{d}t + \eta\,\mathrm{d}W_t^y, \\[0.5em]
\mathrm{d}\langle W^x,W^y\rangle_t
  &= \rho\,\mathrm{d}t.
\end{aligned}
\end{equation*}
