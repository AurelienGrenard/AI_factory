\begin{equation*}
\begin{aligned}
\mathrm{d}r_t
  &= a(b-r_t)\,\mathrm{d}t + \sigma\,\mathrm{d}W_t, \\[0.55em]
\mathbb{E}[r_t\mid r_s]
  &= b + e^{-a\Delta}(r_s-b), \\[0.55em]
\operatorname{Var}(r_t\mid r_s)
  &= \frac{\sigma^2}{2a}\left(1-e^{-2a\Delta}\right),
  \qquad \Delta=t-s.
\end{aligned}
\end{equation*}
