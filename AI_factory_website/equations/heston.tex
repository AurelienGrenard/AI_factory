\begin{equation*}
\begin{aligned}
\mathrm{d}S_t
  &= (r-q)S_t\,\mathrm{d}t
     + \sqrt{v_t}\,S_t\,\mathrm{d}W_t^{S}, \\[0.55em]
\mathrm{d}v_t
  &= \kappa(\theta-v_t)\,\mathrm{d}t
     + \gamma\sqrt{v_t}\,\mathrm{d}W_t^{v}, \\[0.55em]
\mathrm{d}\langle W^{S},W^{v}\rangle_t
  &= \rho\,\mathrm{d}t.
\end{aligned}
\end{equation*}
